## Modelos -------------------------------------------------------------------

# Regresiones logísticas
m1 <- glm(voto ~ educacion_ciudadana,
          family = binomial(link = "logit"), data = did_df)
# educ como control
m2 <- glm(voto ~ educacion_ciudadana + educ,
          family = binomial(link = "logit"), data = did_df)
# edad como control, además de educ. Se usa edad_nac (calculada desde la
# fecha de nacimiento): la edad declarada está desplazada en las olas 85 y 86
m3 <- glm(voto ~ educacion_ciudadana + educ + edad_nac,
          family = binomial(link = "logit"), data = did_df)

# Muestra y modelos para la presentación (index.qmd)
saveRDS(list(datos = did_df, modelos = list(m1, m2, m3)),
        "output/models/modelos_voto.rds")

Este documento carga los datos procesados en `preparation.qmd` y estima los modelos del efecto de la asignatura de Educación Ciudadana sobre el voto en la 2ª vuelta presidencial de 2021 y sobre el interés político. Las tablas se guardan en `output/tables/`, y las muestras y modelos en `output/models/` para presentarlos en la portada.

# Librerías

```{r}
pacman::p_load(dplyr, texreg)
```

# Datos procesados

```{r}
data_voto <- readRDS("input/data/proc/data_voto.rds")
data_int <- readRDS("input/data/proc/data_int.rds")
```

# Asignatura de Educación Ciudadana y voto en 2021

```{r}
## Muestra -------------------------------------------------------------------
did_df <- data_voto %>%
  # se excluye a quienes, según su fecha de nacimiento, no tenían 18 años
  # el día de la 2ª vuelta (19-dic-2021). Quedan fuera también los casos
  # sin fecha de nacimiento.
  dplyr::filter(habilitado_2v == 1) %>%
  dplyr::filter(!is.na(voto), !is.na(educacion_ciudadana))
sjmisc::frq(did_df$voto)
# Tamaño de cada grupo
dplyr::count(did_df, educacion_ciudadana)
dplyr::count(did_df, ola, educacion_ciudadana)

## Modelos -------------------------------------------------------------------

# Regresiones logísticas
m1 <- glm(voto ~ educacion_ciudadana,
          family = binomial(link = "logit"), data = did_df)
# educ como control
m2 <- glm(voto ~ educacion_ciudadana + educ,
          family = binomial(link = "logit"), data = did_df)
# edad como control, además de educ. Se usa edad_nac (calculada desde la
# fecha de nacimiento): la edad declarada está desplazada en las olas 85 y 86
m3 <- glm(voto ~ educacion_ciudadana + educ + edad_nac,
          family = binomial(link = "logit"), data = did_df)

texreg::htmlreg(list(m1, m2, m3), file = "output/tables/modelos_voto.html")

# Muestra y modelos para la presentación (index.qmd)
saveRDS(list(datos = did_df, modelos = list(m1, m2, m3)),
        "output/models/modelos_voto.rds")
```
